          import hashlib
          import json
          import os
          import pathlib
          import statistics
          import subprocess

          root = pathlib.Path(os.environ["RUNNER_TEMP"])
          cpu = min(os.sched_getaffinity(0))
          os.sched_setaffinity(0, {cpu})
          env = os.environ.copy()
          env.pop("PBRS_PROBE_VERIFY", None)
          variants = ("baseline", "candidate")
          programs = {v: root / f"{v}-streaming" for v in variants}
          allocs = {v: root / f"{v}-alloc" for v in variants}
          hashes = {p.name: hashlib.sha256(p.read_bytes()).hexdigest() for p in (*programs.values(), *allocs.values())}
          def run(path, label, shape, mode, messages, calls):
              line = subprocess.check_output([str(path), shape, mode, str(messages), str(calls)],
                                             text=True, timeout=120, env=env).strip()
              parts = line.split()
              assert len(parts) == 7 and parts[0] == label, line
              assert parts[1:5] == [shape, mode, str(messages), str(calls)], line
              return (int(parts[5]), int(parts[6]))
          def measure(variant, shape, mode, messages, calls):
              cpu_ns, wall_ns = run(programs[variant], "PBRS_STREAM_PROBE", shape, mode, messages, calls)
              return {"cpu_ns": cpu_ns, "wall_ns": wall_ns}
          modes = ("none", "bounded", "enabled")
          shapes = ("client", "server", "bidi")
          lengths = {"interactive": (4, 500), "long": (128, 20)}
          for variant in variants:
              for mode in modes:
                  for shape in shapes:
                      for messages, calls in lengths.values():
                          measure(variant, shape, mode, messages, calls)
          results = {}
          for mode in modes:
              for shape in shapes:
                  for size, (messages, calls) in lengths.items():
                      blocks = []
                      for i in range(10):
                          order = (["baseline0", "candidate0", "candidate1", "baseline1"] if i % 2 == 0
                                   else ["candidate0", "baseline0", "baseline1", "candidate1"])
                          rows = {label: measure(label[:-1], shape, mode, messages, calls) for label in order}
                          ratio = ((rows["candidate0"]["cpu_ns"] + rows["candidate1"]["cpu_ns"])
                                   / (rows["baseline0"]["cpu_ns"] + rows["baseline1"]["cpu_ns"]))
                          control_order = ["first", "second"] if i % 2 == 0 else ["second", "first"]
                          controls = {}
                          for variant in variants:
                              values = {label: measure(variant, shape, mode, messages, calls) for label in control_order}
                              controls[variant] = {"order": control_order, "values": values,
                                                   "ratio": values["second"]["cpu_ns"] / values["first"]["cpu_ns"]}
                          blocks.append({"order": order, "runs": rows, "candidate_over_baseline": ratio,
                                         "same_binary_controls": controls})
                      ratios = [block["candidate_over_baseline"] for block in blocks]
                      results[f"{mode}.{shape}.{size}"] = {"messages": messages, "calls": calls, "blocks": blocks,
                                                          "median": statistics.median(ratios),
                                                          "min": min(ratios), "max": max(ratios)}
                      controls = {variant: [block["same_binary_controls"][variant]["ratio"] for block in blocks] for variant in variants}
                      print("CPU", mode, shape, size, f"{statistics.median(ratios):.4f}", f"{min(ratios):.4f}", f"{max(ratios):.4f}",
                            "baseline_control", f"{min(controls['baseline']):.4f}..{max(controls['baseline']):.4f}",
                            "candidate_control", f"{min(controls['candidate']):.4f}..{max(controls['candidate']):.4f}", flush=True)
          allocation_results = {}
          for mode in modes:
              for shape in shapes:
                  samples = []
                  for i in range(5):
                      order = ["baseline", "candidate"] if i % 2 == 0 else ["candidate", "baseline"]
                      measured = {}
                      for variant in order:
                          allocations, requested_bytes = run(allocs[variant], "PBRS_STREAM_ALLOC", shape, mode, 4, 500)
                          assert allocations > 0 and requested_bytes > 0
                          measured[variant] = {"allocations": allocations, "requested_bytes": requested_bytes}
                      samples.append({"order": order, "samples": measured})
                  allocation_results[f"{mode}.{shape}"] = {"messages": 4, "calls": 500, "samples": samples}
                  print("ALLOC", mode, shape, json.dumps(samples, sort_keys=True), flush=True)
          record = {
              "schema": "compact-sender-limits-screen/1",
              "baseline": "c8dd4bd7738074a29dee8727814832a4e754198f", "candidate": "c134820046c26bcbbce9e23c96967bc0f6493631",

              "cpu_results": results, "allocation_results": allocation_results,
              "scope": "isolated compact sender prototype; successful native current-thread client and server share a loopback process and one CPU; no TLS; bounded mode configures a 1024-byte outbound cap; 4-message/500-call and 128-message/20-call streams; 32-byte input/output; ten balanced CPU blocks with same-binary controls; five process-wide allocation pairs per 4-message case count alloc, alloc_zeroed and realloc requests after warmup; no-op log sink in timed enabled mode; no cancellation-event timing; shared host; not a tonic comparison or production ranking"
          }
          path = pathlib.Path("target/streaming-observer-screen.json")

          PY
